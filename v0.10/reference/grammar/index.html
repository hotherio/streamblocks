    def build_grammar(registry: Registry, *, allow_text: bool = True) -> Grammar:
    """Build the grammar of everything ``registry`` can extract.

    The grammar accepts any number of blocks of the registered types, written
    in the registry's syntax. With ``allow_text`` it also accepts lines of free
    text between them, as long as they cannot be mistaken for a block.

    Args:
        registry: Registry holding the syntax and the block types
        allow_text: Allow free text around blocks

    Raises:
        GrammarError: If no block is registered
        NotImplementedError: If the registry's syntax does not support grammars
    """
    if not registry.registered_blocks:
        msg = "Cannot build a grammar: no block type is registered"
        raise GrammarError(msg)

    specs = [block_spec(block_type, block_class) for block_type, block_class in registry.registered_blocks.items()]
    syntax_grammar = registry.syntax.grammar_rules(specs)

    grammar = Grammar()
    alternatives = [Ref(_BLOCK_RULE), Ref(_TEXT_LINE_RULE)]
    grammar.rules[ROOT_RULE] = zero_or_more(choice(*alternatives) if allow_text else Ref(_BLOCK_RULE))
    grammar.rules[_BLOCK_RULE] = choice(*(Ref(_BLOCK_RULE_PREFIX + block_type) for block_type in syntax_grammar.blocks))
    for block_type, node in syntax_grammar.blocks.items():
        grammar.rules[_BLOCK_RULE_PREFIX + block_type] = node
    if allow_text:
        grammar.rules[_TEXT_LINE_RULE] = syntax_grammar.text_line
    grammar.rules.update(syntax_grammar.shared)
    return grammar

    def compile_grammar(registry: Registry, *, dialect: str = GBNF, allow_text: bool = True) -> str:
    """Compile ``registry`` into grammar text in the given dialect.

    Args:
        registry: Registry holding the syntax and the block types
        dialect: Grammar dialect (only ``"gbnf"`` for now)
        allow_text: Allow free text around blocks

    Raises:
        GrammarError: If the dialect is unknown or no block is registered
        NotImplementedError: If the registry's syntax does not support grammars
    """
    renderer = _RENDERERS.get(dialect)
    if renderer is None:
        msg = f"Unknown grammar dialect {dialect!r}; available: {', '.join(_RENDERERS)}"
        raise GrammarError(msg)
    return renderer(build_grammar(registry, allow_text=allow_text))

    def block_spec(block_type: str, block_class: type[Block[Any, Any]]) -> BlockSpec:
    """Describe ``block_class``'s metadata fields from its JSON schema."""
    metadata_class, content_class = extract_block_types(block_class)
    schema = metadata_class.model_json_schema()
    required = set(cast("list[str]", schema.get("required", [])))
    properties = cast("dict[str, dict[str, Any]]", schema.get("properties", {}))
    fields = tuple(
        _metadata_field(name, definition, required=name in required)
        for name, definition in properties.items()
        if name not in {_ID_FIELD, _BLOCK_TYPE_FIELD}
    )
    return BlockSpec(block_type=block_type, fields=fields, content_class=content_class)

    class SharedLines:
    """Content-line rules for a syntax, defined once and referred to by name.

    Calling the instance returns a reference to a rule matching a full line
    that starts with none of the syntax ``markers`` nor the extra prefixes.
    The rules themselves accumulate in :attr:`rules`.
    """

    def __init__(self, *markers: str, name: str = "line") -> None:
        """Create a factory for lines that avoid ``markers``."""
        self._markers = markers
        self._name = name
        self._names: dict[tuple[tuple[str, ...], bool], str] = {}
        self.rules: dict[str, Node] = {}

    def __call__(self, *prefixes: str, non_blank: bool = False) -> Node:
        """Return a reference to the line rule avoiding the markers and ``prefixes``."""
        key = (prefixes, non_blank)
        if key not in self._names:
            rule = f"{self._name}-{len(self._names)}" if self._names else self._name
            self._names[key] = rule
            self.rules[rule] = line_not_starting_with(*self._markers, *prefixes, non_blank=non_blank)
        return Ref(self._names[key])

    def render_gbnf(grammar: Grammar) -> str:
    """Render ``grammar`` as GBNF text, root rule first."""
    names = [ROOT_RULE, *(name for name in grammar.rules if name != ROOT_RULE)]
    return "\n".join(f"{rule_name(name)} ::= {_render(grammar.rules[name])}" for name in names) + "\n"
